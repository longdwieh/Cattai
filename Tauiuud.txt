-- Function to create toggle options
local function createOption(y, name, desc)
	local frame = Instance.new("Frame", content)
	frame.Size = UDim2.new(1, -20, 0, 60)
	frame.Position = UDim2.new(0, 0, 0, y)
	frame.BackgroundColor3 = Color3.fromRGB(25, 45, 30)
	frame.BorderSizePixel = 0
	Instance.new("UICorner", frame).CornerRadius = UDim.new(0, 8)

	local title = Instance.new("TextLabel", frame)
	title.Text = name
	title.Font = Enum.Font.GothamSemibold
	title.TextSize = 18
	title.TextColor3 = Color3.fromRGB(230, 230, 230)
	title.BackgroundTransparency = 1
	title.Size = UDim2.new(0.7, 0, 0, 25)
	title.Position = UDim2.new(0, 12, 0, 8)
	title.TextXAlignment = Enum.TextXAlignment.Left

	local descText = Instance.new("TextLabel", frame)
	descText.Text = desc
	descText.Font = Enum.Font.Gotham
	descText.TextSize = 14
	descText.TextColor3 = Color3.fromRGB(160, 160, 160)
	descText.BackgroundTransparency = 1
	descText.Size = UDim2.new(0.9, 0, 0, 18)
	descText.Position = UDim2.new(0, 12, 0, 34)
	descText.TextXAlignment = Enum.TextXAlignment.Left

	local toggle = Instance.new("Frame", frame)
	toggle.Size = UDim2.new(0, 60, 0, 30)
	toggle.Position = UDim2.new(1, -80, 0.5, -15)
	toggle.BackgroundColor3 = Color3.fromRGB(15, 25, 20)
	Instance.new("UICorner", toggle).CornerRadius = UDim.new(0, 15)

	local knob = Instance.new("Frame", toggle)
	knob.Size = UDim2.new(0, 26, 0, 26)
	knob.Position = UDim2.new(1, -28, 0.5, -13)
	knob.BackgroundColor3 = Color3.fromRGB(0, 170, 255)
	Instance.new("UICorner", knob).CornerRadius = UDim.new(1, 0)
end

createOption(60, "Auto Farm Level", "Autofarms Quests and Mobs")
createOption(130, "Farm Nearest Mob", "Autofarms nearest mobs")
createOption(200, "Include Bosses", "Include bosses in routine")

-- 🟢 Toggle visibility
local isVisible = true
avatarBtn.MouseButton1Click:Connect(function()
	isVisible = not isVisible
	main.Visible = isVisible
end)
